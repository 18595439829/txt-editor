### JavaScript改动分析

**菜单显示函数**（第461-469行）
```javascript
function showFileContextMenu(e, fileId) {
  const menu = document.getElementById("fileContextMenu");
  menu.classList.add("show");                    // 显示菜单
  menu.style.left = e.clientX + "px";           // 设置X位置
  menu.style.top = e.clientY + "px";            // 设置Y位置
  menu.dataset.currentFileId = fileId;          // 保存文件ID
}
```

**菜单项处理**（第478-511行）
```javascript
// 导出为JSON
document.getElementById("exportJsonBtn").addEventListener("click", async () => {
  const menu = document.getElementById("fileContextMenu");
  const fileId = menu.dataset.currentFileId;
  await exporter.exportSingleFileAsJson(fileId, true);
  hideFileContextMenu();
});

// 导出为ZIP
document.getElementById("exportZipBtn").addEventListener("click", async () => {
  const menu = document.getElementById("fileContextMenu");
  const fileId = menu.dataset.currentFileId;
  await exporter.exportSingleFileAsZip(fileId, true);
  hideFileContextMenu();
});

// 导入文件
document.getElementById("importFileBtn").addEventListener("click", () => {
  const fileInput = document.createElement("input");
  fileInput.type = "file";
  fileInput.accept = ".json,.zip";
  fileInput.onchange = async (e) => {
    const result = await importer.importSingleFile(e.target.files[0]);
    // 处理导入结果
  };
  fileInput.click();
});
```
